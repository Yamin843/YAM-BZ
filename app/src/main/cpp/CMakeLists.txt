cmake_minimum_required(VERSION 3.22.1)
project("yambz" CXX C)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# ─────────────────────────────────────────────────────────────
#  ImGui
# ─────────────────────────────────────────────────────────────
set(IMGUI_DIR ${CMAKE_CURRENT_SOURCE_DIR}/imgui)

if(NOT EXISTS ${IMGUI_DIR}/imgui.cpp)
    message(FATAL_ERROR "ImGui missing — run scripts/fetch_imgui.sh first")
endif()

add_library(imgui STATIC
    ${IMGUI_DIR}/imgui.cpp
    ${IMGUI_DIR}/imgui_draw.cpp
    ${IMGUI_DIR}/imgui_tables.cpp
    ${IMGUI_DIR}/imgui_widgets.cpp
    ${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp
)
target_include_directories(imgui PUBLIC
    ${IMGUI_DIR}
    ${IMGUI_DIR}/backends
)
target_compile_definitions(imgui PUBLIC
    IMGUI_IMPL_OPENGL_ES3
    IMGUI_DEFINE_MATH_OPERATORS
)
target_link_libraries(imgui PUBLIC GLESv3 EGL android log)

# ─────────────────────────────────────────────────────────────
#  Albatross header (declarations only — actual .so embedded at runtime)
# ─────────────────────────────────────────────────────────────
set(ALBATROSS_INCLUDE ${CMAKE_CURRENT_SOURCE_DIR}/../../../../albatross-libs/include)

if(NOT EXISTS ${ALBATROSS_INCLUDE}/albatross.h)
    message(FATAL_ERROR "albatross.h missing at ${ALBATROSS_INCLUDE}")
endif()

# ─────────────────────────────────────────────────────────────
#  YAMBZ main library — SINGLE .so
# ─────────────────────────────────────────────────────────────
file(GLOB_RECURSE YAMBZ_CPP
    ${CMAKE_CURRENT_SOURCE_DIR}/yambz/*.cpp
)

add_library(yambz SHARED ${YAMBZ_CPP})

target_include_directories(yambz PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/yambz
    ${ALBATROSS_INCLUDE}
    ${IMGUI_DIR}
    ${IMGUI_DIR}/backends
)

target_compile_definitions(yambz PRIVATE
    YAMBZ_VERSION="1.0.0"
    YAMBZ_LOG_TAG="YAMBZ"
    IMGUI_DEFINE_MATH_OPERATORS
)

target_link_libraries(yambz
    imgui
    android
    log
    GLESv3
    EGL
    dl
)

# 16KB page alignment for Android 15+
target_link_options(yambz PRIVATE -Wl,-z,max-page-size=16384)
target_link_options(yambz PRIVATE -Wl,--no-undefined)
target_link_options(yambz PRIVATE -Wl,--exclude-libs,ALL)

# Reduce binary size
target_compile_options(yambz PRIVATE
    $<$<CONFIG:Release>:-Os>
    -fvisibility=hidden
    -fvisibility-inlines-hidden
)

# Verify embedded headers exist (generated by scripts/embed_assets.sh)
set(EMBED_DEX_H    ${CMAKE_CURRENT_SOURCE_DIR}/yambz/embed/embedded_dex.h)
set(EMBED_ALB_H    ${CMAKE_CURRENT_SOURCE_DIR}/yambz/embed/embedded_albatross.h)

if(EXISTS ${EMBED_DEX_H} AND EXISTS ${EMBED_ALB_H})
    message(STATUS "YAMBZ: embedded dex + albatross enabled")
    target_compile_definitions(yambz PRIVATE YAMBZ_HAS_EMBEDDED=1)
else()
    message(WARNING "YAMBZ: embedded assets NOT found (build will fail unless provided)")
    target_compile_definitions(yambz PRIVATE YAMBZ_HAS_EMBEDDED=0)
endif()
